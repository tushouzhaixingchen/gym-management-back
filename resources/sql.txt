-- ==================== 1. 门店表 ====================
CREATE TABLE `stores` (
    `id` INT PRIMARY KEY AUTO_INCREMENT COMMENT '门店ID',
    `store_name` VARCHAR(100) NOT NULL COMMENT '门店名称',
    `store_code` VARCHAR(50) UNIQUE NOT NULL COMMENT '门店编码',
    `province` VARCHAR(50) COMMENT '省份',
    `city` VARCHAR(50) COMMENT '城市',
    `district` VARCHAR(50) COMMENT '区域',
    `address` VARCHAR(255) COMMENT '详细地址',
    `phone` VARCHAR(20) COMMENT '联系电话',
    `manager_name` VARCHAR(50) COMMENT '店长姓名',
    `manager_phone` VARCHAR(20) COMMENT '店长电话',
    `open_time` TIME COMMENT '营业时间-开始',
    `close_time` TIME COMMENT '营业时间-结束',
    `status` TINYINT DEFAULT 1 COMMENT '1营业 0停业 2装修中',
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    INDEX `idx_status` (`status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='门店表';

-- ==================== 4. 管理员表 ====================
CREATE TABLE `admins` (
    `id` INT PRIMARY KEY AUTO_INCREMENT COMMENT '管理员ID',
    `username` VARCHAR(50) NOT NULL UNIQUE COMMENT '登录账号',
    `password_hash` VARCHAR(255) NOT NULL COMMENT '密码哈希',
    `real_name` VARCHAR(50) COMMENT '真实姓名',
    `role_id` INT NOT NULL COMMENT '角色ID',
    `store_id` INT DEFAULT NULL COMMENT '所属门店ID（超级管理员为NULL）',
    `phone` VARCHAR(20) COMMENT '联系电话',
    `email` VARCHAR(100) COMMENT '邮箱',
    `status` TINYINT DEFAULT 1 COMMENT '1正常 0禁用',
    `last_login_at` DATETIME COMMENT '最后登录时间',
    `last_login_ip` VARCHAR(50) COMMENT '最后登录IP',
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    FOREIGN KEY (`role_id`) REFERENCES `roles`(`id`),
    INDEX `idx_store_id` (`store_id`),
    INDEX `idx_status` (`status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='管理员表';

-- ==================== 5. 角色权限关联表 ====================
CREATE TABLE `role_permissions` (
    `role_id` INT NOT NULL COMMENT '角色ID',
    `permission_id` INT NOT NULL COMMENT '权限ID',
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (`role_id`, `permission_id`),
    FOREIGN KEY (`role_id`) REFERENCES `roles`(`id`) ON DELETE CASCADE,
    FOREIGN KEY (`permission_id`) REFERENCES `permissions`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='角色权限关联表';

-- 超级管理员拥有所有权限
INSERT INTO `role_permissions` (`role_id`, `permission_id`)
SELECT 1, `id` FROM `permissions`;

-- 门店管理员权限（排除跨店权限）
INSERT INTO `role_permissions` (`role_id`, `permission_id`)
SELECT 2, `id` FROM `permissions`
WHERE `permission_code` NOT IN ('admin:create', 'admin:delete', 'report:all_stores');

-- ==================== 6. 会员表 ====================
CREATE TABLE `members` (
    `id` INT PRIMARY KEY AUTO_INCREMENT COMMENT '会员ID',
    `member_no` VARCHAR(50) UNIQUE NOT NULL COMMENT '会员卡号（全系统唯一）',
    `real_name` VARCHAR(50) NOT NULL COMMENT '真实姓名',
    `gender` TINYINT DEFAULT 0 COMMENT '性别：0未知 1男 2女',
    `phone` VARCHAR(20) NOT NULL COMMENT '联系电话',
    `email` VARCHAR(100) COMMENT '邮箱',
    `birthday` DATE COMMENT '出生日期',
    `register_store_id` INT COMMENT '注册门店ID（首次办卡门店）',
    `join_date` DATE NOT NULL COMMENT '入会日期',
    `expire_date` DATE COMMENT '会员到期日期',members
    `card_type` VARCHAR(20) DEFAULT 'period' COMMENT '卡类型：period(期限卡)/times(次卡)/vip(贵宾卡)',
    `total_times` INT DEFAULT 0 COMMENT '总次数（次卡用）',
    `remaining_times` INT DEFAULT 0 COMMENT '剩余次数（次卡用）',
    `balance` DECIMAL(10,2) DEFAULT 0 COMMENT '账户余额',
    `total_consumption` DECIMAL(10,2) DEFAULT 0 COMMENT '累计消费金额',
    `visit_count` INT DEFAULT 0 COMMENT '总到店次数',
    `last_visit_at` DATETIME COMMENT '最后到店时间',
    `last_visit_store_id` INT COMMENT '最后到店门店ID',
    `status` TINYINT DEFAULT 1 COMMENT '1正常 0过期 2冻结',
    `remark` TEXT COMMENT '备注',
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    FOREIGN KEY (`register_store_id`) REFERENCES `stores`(`id`),
    FOREIGN KEY (`last_visit_store_id`) REFERENCES `stores`(`id`),
    INDEX `idx_member_no` (`member_no`),
    INDEX `idx_phone` (`phone`),
    INDEX `idx_expire_date` (`expire_date`),
    INDEX `idx_status` (`status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='会员表';

-- ==================== 7. 会员到店记录表（简化：只记人次）====================
CREATE TABLE `member_visits` (
    `id` INT PRIMARY KEY AUTO_INCREMENT COMMENT '到店记录ID',
    `member_id` INT NOT NULL COMMENT '会员ID',
    `store_id` INT NOT NULL COMMENT '到店门店ID',
    `visit_date` DATE NOT NULL COMMENT '到店日期',
    `check_in_time` DATETIME NOT NULL COMMENT '签到时间',
    `check_in_method` VARCHAR(20) DEFAULT 'manual' COMMENT '签到方式：manual(手动)/card(刷卡)/qrcode(二维码)/face(人脸)/device(设备)',
    `card_type` VARCHAR(20) DEFAULT 'period' COMMENT '当时卡类型：period/times/vip',
    `deduct_times` INT DEFAULT 0 COMMENT '扣除次数（次卡用，期限卡为0）',
    `related_appointment_id` INT DEFAULT NULL COMMENT '关联预约ID（有私教预约则填）',
    `related_course_id` INT DEFAULT NULL COMMENT '关联课程ID（有团课预约则填）',
    `staff_id` INT DEFAULT NULL COMMENT '登记员工ID',
    `remark` TEXT COMMENT '备注',
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    FOREIGN KEY (`member_id`) REFERENCES `members`(`id`),
    FOREIGN KEY (`store_id`) REFERENCES `stores`(`id`),
    FOREIGN KEY (`related_appointment_id`) REFERENCES `appointments`(`id`),
    FOREIGN KEY (`related_course_id`) REFERENCES `courses`(`id`),
    FOREIGN KEY (`staff_id`) REFERENCES `employees`(`id`),
    INDEX `idx_member_date` (`member_id`, `visit_date`),
    INDEX `idx_store_date` (`store_id`, `visit_date`),
    INDEX `idx_visit_date` (`visit_date`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='会员到店记录表';

    `store_id` INT NOT NULL COMMENT '所属门店ID',
    `equipment_id` INT NOT NULL COMMENT '器械ID',
    `maintenance_no` VARCHAR(50) UNIQUE NOT NULL COMMENT '维护单号',
    `maintenance_type` VARCHAR(20) NOT NULL COMMENT '维护类型：定期/故障/大修',
    `maintenance_date` DATE NOT NULL COMMENT '维护日期',
    `maintenance_staff` VARCHAR(50) COMMENT '维护人员',
    `maintenance_cost` DECIMAL(10,2) DEFAULT 0 COMMENT '维护费用',
    `description` TEXT COMMENT '维护内容描述',
    `status` TINYINT DEFAULT 1 COMMENT '0待处理 1处理中 2已完成',
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    FOREIGN KEY (`store_id`) REFERENCES `stores`(`id`),
    FOREIGN KEY (`equipment_id`) REFERENCES `equipments`(`id`),
    INDEX `idx_equipment_date` (`equipment_id`, `maintenance_date`),
    INDEX `idx_store_date` (`store_id`, `maintenance_date`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='器械维护记录表';

-- ==================== 17. 订单表 ====================
CREATE TABLE `orders` (
    `id` INT PRIMARY KEY AUTO_INCREMENT COMMENT '订单ID',
    `store_id` INT NOT NULL COMMENT '消费门店ID',
    `order_no` VARCHAR(50) UNIQUE NOT NULL COMMENT '订单号',
    `member_id` INT NOT NULL COMMENT '会员ID',
    `order_type` VARCHAR(20) NOT NULL COMMENT '订单类型：membership/course/appointment/product',
    `order_amount` DECIMAL(10,2) NOT NULL COMMENT '订单金额',
    `discount_amount` DECIMAL(10,2) DEFAULT 0 COMMENT '优惠金额',
    `pay_amount` DECIMAL(10,2) NOT NULL COMMENT '实付金额',
    `pay_method` VARCHAR(20) COMMENT '支付方式：wechat/alipay/cash/card',
    `pay_status` TINYINT DEFAULT 0 COMMENT '0未支付 1已支付 2已退款',
    `pay_time` DATETIME COMMENT '支付时间',
    `sales_employee_id` INT COMMENT '销售员工ID',
    `remark` TEXT COMMENT '备注',
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    FOREIGN KEY (`store_id`) REFERENCES `stores`(`id`),
    FOREIGN KEY (`member_id`) REFERENCES `members`(`id`),
    FOREIGN KEY (`sales_employee_id`) REFERENCES `employees`(`id`),
    INDEX `idx_store_time` (`store_id`, `created_at`),
    INDEX `idx_pay_status` (`pay_status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='订单表';
