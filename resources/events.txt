-- 创建新的事件
CREATE EVENT auto_end_courses
ON SCHEDULE EVERY 1 MINUTE  -- 设定频率为每1分钟执行一次
STARTS CURRENT_TIMESTAMP    -- 从当前时间开始生效
ENABLE                      -- 创建后直接启用
DO
  UPDATE courses
  SET status = 4            -- 将状态更新为 4 (已结束)
  WHERE end_time < NOW()    -- 筛选结束时间早于当前时间的课程
    AND status != 4;        -- 排除已经是“已结束”状态的，提升性能

-- 2. 创建事件
CREATE EVENT IF NOT EXISTS `evt_update_booking_status_when_course_ends`
ON SCHEDULE EVERY 1 MINUTE
STARTS CURRENT_TIMESTAMP
ON COMPLETION PRESERVE
ENABLE
COMMENT '每分钟扫描已结束的课程，并将对应的报名记录状态更新为已结束'
DO
BEGIN
    -- 核心逻辑：更新 course_bookings 表
    -- 逻辑说明：
    -- 1. 关联 courses 表
    -- 2. 筛选条件：课程状态为 4(已结束) AND 报名状态为 0(已报名/未结束)
    UPDATE `course_bookings` AS cb
    JOIN `courses` AS c ON cb.`course_id` = c.`id`
    SET 
        cb.`status` = 1,                -- 将报名状态改为 1 (已结束)
        cb.`updated_at` = CURRENT_TIMESTAMP -- 更新时间戳
    WHERE 
        c.`status` = 4                  -- 课程状态：4 = 已结束
        AND cb.`status` = 0;            -- 只更新当前还是“已报名”状态的记录，避免重复更新
        
    -- 注意：如果你不需要记录日志，上面的 UPDATE 语句执行完即可。
    -- 如果你确实需要获取影响行数用于调试，请确保你的 MySQL 版本支持 GET DIAGNOSTICS
    -- 在大多数简单场景下，直接执行 UPDATE 即可，无需额外获取行数。
END$$
